# Dependencias del backend del Módulo RRHH — deliberadamente mínimas.
#
# flask    — sirve la API y el propio .dc.html en el mismo origen (sin CORS)
# requests — cliente HTTP hacia yunatt.com; su urllib3 es el que permite
#            forzar TLS 1.3, requisito del handshake con yunatt en Windows
#
# El almacenamiento usa sqlite3, que viene en la librería estándar de
# Python — no hay servidor de base de datos que instalar ni administrar.

# Versiones mínimas, no fijas: este equipo ya tiene flask y requests
# instalados de forma global. Fijar versiones exactas forzaría un
# downgrade que afectaría también al ERP anterior mientras siga en
# producción, porque comparten el mismo Python del sistema.
flask>=3.0
requests>=2.31
